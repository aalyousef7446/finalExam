    <div class="stats">

        <button class="btn-add" onclick="addEmployee()">إضافة</button>

<script>
    // مصفوفة لتخزين الموظفات محلياً
    let employees = JSON.parse(localStorage.getItem('attendance_list')) || [];

    // دالة لتحديث الواجهة والإحصائيات
    function updateUI() {
        const list = document.getElementById('employee-list');
        list.innerHTML = '';
        
        let present = 0;
        let absent = 0;

        employees.forEach((emp, index) => {
            if(emp.status === 'حاضر') present++;
            if(emp.status === 'غائب') absent++;

            const row = document.createElement('tr');
            row.innerHTML = `
                <td><strong>${emp.name}</strong></td>
                <td>
                    <button class="status-btn btn-present ${emp.status === 'حاضر' ? 'active' : ''}" onclick="toggleStatus(${index}, 'حاضر')">حاضر</button>
                    <button class="status-btn btn-absent ${emp.status === 'غائب' ? 'active' : ''}" onclick="toggleStatus(${index}, 'غائب')">غائب</button>
                </td>
                <td>
                    <button style="background: none; border: none; color: #e74c3c; cursor: pointer;" onclick="deleteEmployee(${index})">❌ حذف</button>
                </td>
            `;
            list.appendChild(row);
        });

        document.getElementById('total-count').innerText = employees.length;
        document.getElementById('present-count').innerText = present;
        document.getElementById('absent-count').innerText = absent;
        
        localStorage.setItem('attendance_list', JSON.stringify(employees));
    }

    // دالة إضافة موظفة جديدة
    function addEmployee() {
        const nameInput = document.getElementById('emp-name');
        const name = nameInput.value.trim();
        
        if (name === '') {
            alert('الرجاء إدخال اسم الموظفة');
            return;
        }

        employees.push({ name: name, status: 'غائب' }); // الحالة الافتراضية غائب حتى يتم تحضيرها
        nameInput.value = '';
        updateUI();
    }

    // دالة تغيير حالة الحضور
    function toggleStatus(index, status) {
        employees[index].status = status;
        updateUI();
    }

    // دالة حذف موظفة
    function deleteEmployee(index) {
        if(confirm('هل أنت متأكد من حذف هذه الموظفة؟')) {
            employees.splice(index, 1);
            updateUI();
        }
    }

    // دالة تصدير البيانات إلى ملف Excel/CSV
    function exportToCSV() {
        if(employees.length === 0) {
            alert('لا توجد بيانات لتصديرها');
            return;
        }

        let csvContent = "\uFEFF"; // لدعم اللغة العربية في إكسل
        csvContent += "اسم الموظفة,حالة حضور الطابور\n";
        
        employees.forEach(emp => {
            csvContent += `${emp.name},${emp.status}\n`;
        });

        const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
        const link = document.createElement("a");
        const url = URL.createObjectURL(blob);
        
        const date = new Date().toLocaleDateString('ar-EG').replace(/\//g, '-');
        link.setAttribute("href", url);
        link.setAttribute("download", `حضور_الطابور_${date}.csv`);
        link.style.visibility = 'hidden';
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    }

    // تشغيل التحديث الأولي عند فتح الصفحة
    updateUI();
</script>
